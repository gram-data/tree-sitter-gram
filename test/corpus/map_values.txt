==================
Map values are non-nested maps, similar in syntax to records but only accepting scalar values.
==================

{
  address: {
    street: "123 Main St",
    city: "Anytown",
    state: "CA",
    zip: "12345"
  }
}

---

(gram_pattern
  root: (record
    (record_property
      key: (symbol)
      value: (map
        (map_entry
          key: (symbol)
          value: (string_literal
            content: (string_content)))
        (map_entry
          key: (symbol)
          value: (string_literal
            content: (string_content)))
        (map_entry
          key: (symbol)
          value: (string_literal
            content: (string_content)))
        (map_entry
          key: (symbol)
          value: (string_literal
            content: (string_content)))))))

==================
Map values may use quoted keys
==================

(a:Person {
  address: {
    `street name`: "123 Main St",
    "postal code": "12345",
    city: "Anytown"
  }
})

---

(gram_pattern
  (node_pattern
      identifier: (symbol)
      labels: (labels
        (symbol))
      record: (record
        (record_property
          key: (symbol)
          value: (map
            (map_entry
              key: (quoted_name
                content: (string_content))
              value: (string_literal
                content: (string_content)))
            (map_entry
              key: (double_quoted_name
                content: (string_content))
              value: (string_literal
                content: (string_content)))
            (map_entry
              key: (symbol)
              value: (string_literal
                content: (string_content))))))))

==================
Map values may not be nested
:error
==================

{
  address: {
    street: {
      number: "123",
      name: "Main St"
    },
    city: "Anytown"
  }
}

---

==================
Map values may not appear within an array
:error
==================

(a:Person {
  addresses [{
    street: "123 Main St",
    city: "Anytown",
    state: "CA",
    zip: "12345"
  }]
})

---

==================
Map values may contain arrays of scalars
==================

(a:Person {
  identity: {
    name: "John Doe",
    aliases: ["John", "Johnny"]
  }
})

---

(gram_pattern
  (node_pattern
    (symbol)

      (symbol))
    (record
      (record_property
        (symbol)
        (map
          (map_entry
            (symbol)
            (string_literal
              (string_content)))
          (map_entry
            (symbol)
            (array
              (string_literal
                (string_content))
              (string_literal
                (string_content)))))))))

==================
Map values may not contain arrays of maps
:error
==================

(a:Person {
  identity: {
    aliases: [{ name: "John" }]
  }
})

---

==================
Map values may not contain nested arrays
:error
==================

(a:Person {
  identity: {
    aliases: [["John"]]
  }
})
